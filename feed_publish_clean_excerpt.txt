# D259 served-stamp verification (instrumented C build, 2026-10-26):
# after a successful PATCH, the publisher polls the SERVED raw payload
# of each live-pick file for the feed_ts it just published. The 52.0s
# deadline is the priced value from the settling tests (false-fire at
# 52s pooled 5/131 = 3.82%, Wilson upper 8.62% < 10%); elapsed time is
# compared at full float precision — never rounded to whole seconds.
# A stamp still unserved at the deadline earns exactly ONE re-PATCH
# (a byte-identical PATCH still creates a new gist revision —
# SETTLED-PASS, 2026-10-26), then a bounded confirmation poll.
VERIFY_SERVED_DEADLINE_SEC = 52.0
VERIFY_SERVED_POLL_SEC = 4.0
VERIFY_SERVED_CONFIRM_SEC = 15.0


def live_feed_files(live):
    """Sorted feed file names for the sports in a live_pick_games()
    list (entries "SPORT: matchup"). Unknown sports are ignored —
    they cannot name a feed file, so there is nothing to verify."""
    out = set()
    for entry in live or []:
        sport = str(entry).split(":", 1)[0].strip().upper()
        name = FEED_FILE_BY_SPORT.get(sport)
        if name:
            out.add(name)
    return sorted(out)


def published_feed_ts(files, filename):
    """The feed_ts a build_files() result carries for one file, or
    None when the content is unparseable / carries no stamp — such a
    file has no stamp to verify and is left out of verification."""
    entry = (files or {}).get(filename)
    if entry is None:
        return None
    content = entry["content"] if isinstance(entry, dict) else str(entry)
    try:
        payload = json.loads(content)
    except Exception:
        return None
    if isinstance(payload, dict):
        return payload.get("feed_ts")
    return None


def fetch_served_payload(raw_base, filename):
    """The SERVED payload of one gist raw file, cache-busted the way
    the page fetches it (?t=<now>). Uses fetch() (curl); any failure
    raises, and the caller treats a failed read as not-served."""
    bust = int(time.time() * 1000)
    return fetch(f"{raw_base}/{filename}?t={bust}")
